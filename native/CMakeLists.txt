cmake_minimum_required(VERSION 3.28)
project(mifitness_bridge LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(DEPS "${ROOT}/third_party")
foreach(dep IN ITEMS "lsplant/lsplant/src/main/jni/CMakeLists.txt" "dobby/CMakeLists.txt" "xdl/xdl/src/main/cpp/CMakeLists.txt")
  if(NOT EXISTS "${DEPS}/${dep}")
    message(FATAL_ERROR "Missing dependency ${dep}. Run scripts/fetch-native-deps.sh")
  endif()
endforeach()
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
set(Plugin.ImportTableReplace OFF CACHE BOOL "" FORCE)
set(Plugin.Android.BionicLinkerUtil OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/dobby" "${CMAKE_BINARY_DIR}/dobby")
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(DEBUG_SYMBOLS_PATH "${CMAKE_BINARY_DIR}/symbols" CACHE PATH "" FORCE)
add_subdirectory("${DEPS}/lsplant/lsplant/src/main/jni" "${CMAKE_BINARY_DIR}/lsplant")
file(GLOB XDL_SOURCES "${DEPS}/xdl/xdl/src/main/cpp/*.c")
add_library(xdl_static STATIC ${XDL_SOURCES})
target_compile_features(xdl_static PRIVATE c_std_17)
target_include_directories(xdl_static PUBLIC
    "${DEPS}/xdl/xdl/src/main/cpp/include"
    "${DEPS}/xdl/xdl/src/main/cpp")
add_library(mifitness_bridge SHARED src/entry.cpp)
target_include_directories(mifitness_bridge PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/include"
    "${DEPS}/lsplant/lsplant/src/main/jni/include")
target_compile_options(mifitness_bridge PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden -fno-exceptions -fno-rtti)
target_link_options(mifitness_bridge PRIVATE "-Wl,-z,relro,-z,now" "-Wl,-z,max-page-size=16384")
target_link_libraries(mifitness_bridge PRIVATE dobby lsplant_static dex_builder_static xdl_static log dl)
